#!/bin/bash
echo "Enter the username"
read us
for ((i=1;i<=10;i++))
do
useradd $us$i
echo "Below are details of user $us$i"
grep $us$i /etc/passwd
echo "$us$i" |passwd "$us$i" --stdin
echo "Below are details of password set or not for user $us$i"
grep "$us$i" /etc/shadow
echo "========================================"
done
